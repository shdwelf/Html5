import ghidra.app.util.bin.ByteProvider;
import ghidra.app.util.importer.MessageLog;

import ghidra.app.util.opinion.Loader;

/**
 * DOS Binary Loader for Ghidra
 * 
 * Handles three formats:
 *   1. MZ .EXE — DOS MZ executables with relocation table
 *   2. COM     — Flat binaries loaded at CS:0100 (with PSP)
 *   3. RAW     — Flat binaries loaded at CS:0000
 * 
 * Maps to Ghidra language: x86:LE:16:Real Mode
 */
public class DOSBinaryLoader implements Loader {

    private static final LanguageID LANG_ID = new LanguageID("x86:LE:16:Real Mode");
    
    // MZ Header constants
    private static final int MZ_MAGIC       = 0x00; // "MZ" or "ZM"
    private static final int MZ_LAST_PAGE   = 0x02; // Bytes on last page
    private static final int MZ_PAGES       = 0x04; // Total pages (512 bytes)
    private static final int MZ_RELOCS      = 0x06; // Relocation entries
    private static final int MZ_HDR_SIZE    = 0x08; // Header size in paragraphs
    private static final int MZ_MIN_ALLOC   = 0x0A; // Min memory (paragraphs)
    private static final int MZ_MAX_ALLOC   = 0x0C; // Max memory (paragraphs)
    private static final int MZ_SS          = 0x0E; // Initial SS
    private static final int MZ_SP          = 0x10; // Initial SP
    private static final int MZ_CHECKSUM    = 0x12; // Checksum
    private static final int MZ_IP          = 0x14; // Initial IP
    private static final int MZ_CS          = 0x16; // Initial CS
    private static final int MZ_RELOC_OFF   = 0x18; // Relocation table offset
    private static final int MZ_OVERLAY     = 0x1C; // Overlay number
    
    // COM constants
    private static final int COM_LOAD_SEG   = 0x1000; // Typical CS for COM
    private static final int PSP_SIZE       = 0x100;   // 256-byte PSP

    // ─── Format Detection ─────────────────────────────────────

    public enum DosFormat {
        MZ_EXE,  // MZ executable with header
        COM,     // Flat .COM (size ≤ 64KB - PSP)
        RAW      // Raw binary (fallback)
    }

    private DosFormat detectFormat(ByteProvider provider) throws IOException {
        long size = provider.length();
        if (size < 2) return DosFormat.RAW;

        byte[] magic = provider.readBytes(0, 2);
        
        // Check MZ signature
        if ((magic[0] == 'M' && magic[1] == 'Z') ||
            (magic[0] == 'Z' && magic[1] == 'M')) {
            return DosFormat.MZ_EXE;
        }
        
        // COM: must fit in 64KB minus PSP
        if (size <= 0xFF00) {
            return DosFormat.COM;
        }
        
        return DosFormat.RAW;
    }

    // ─── Load Spec ────────────────────────────────────────────

    @Override
    public Collection<LoadSpec> findSupportedLoadSpecs(ByteProvider provider)
            throws IOException {
        
        List<LoadSpec> specs = new ArrayList<>();
        DosFormat fmt = detectFormat(provider);
        
        if (fmt != DosFormat.RAW) {
            specs.add(new LoadSpec(this, 0, 
                new LanguageCompilerSpecPair(LANG_ID, "default"), true));
        }
        
        return specs;
    }

    @Override
    public String getName() {
        return "DOS Binary Loader (MZ / COM / Raw)";
    }

    // ─── Loader Logic ─────────────────────────────────────────

    @Override
    public List<LinkedOption> getDefaultOptions(
            ByteProvider provider, DomainObject domainObject,
            LoadSpec loadSpec, Object loadData) {
        return new ArrayList<>();
    }

    @Override
    public String validateOptions(ByteProvider provider,
            DomainObject domainObj, LoadSpec loadSpec,
            List<Option> options, Object loadData) {
        return null; // no validation issues
    }

    @Override
    protected List<LoadResults<Program>> load(
            ByteProvider provider, String preferredNamespace,
            DomainObject domainObj, LoadSpec loadSpec,
            List<Option> options, MessageLog log, Object consumer,
            TaskMonitor monitor) throws IOException, CancelledException {

        DosFormat fmt = detectFormat(provider);
        Program program = (Program) domainObj;
        Memory memory = program.getMemory();
        AddressFactory addrFactory = program.getAddressFactory();
        
        switch (fmt) {
            case MZ_EXE:
                loadMZ(provider, memory, addrFactory, program, log, monitor);
                break;
            case COM:
                loadCOM(provider, memory, addrFactory, program, log, monitor);
                break;
            default:
                loadRAW(provider, memory, addrFactory, program, log, monitor);
                break;
        }
        
        return Collections.emptyList();
    }

    // ─── MZ EXE Loader ────────────────────────────────────────

    private void loadMZ(ByteProvider provider, Memory memory,
            AddressFactory addrFactory, Program program,
            MessageLog log, TaskMonitor monitor) throws IOException {

        byte[] header = provider.readBytes(0, 0x1C + 4);
        
        int lastPage    = readLE16(header, MZ_LAST_PAGE);
        int pages       = readLE16(header, MZ_PAGES);
        int relocCount  = readLE16(header, MZ_RELOCS);
        int hdrParas    = readLE16(header, MZ_HDR_SIZE);
        int minAlloc    = readLE16(header, MZ_MIN_ALLOC);
        int maxAlloc    = readLE16(header, MZ_MAX_ALLOC);
        int initSS      = readLE16(header, MZ_SS);
        int initSP      = readLE16(header, MZ_SP);
        int initIP      = readLE16(header, MZ_IP);
        int initCS      = readLE16(header, MZ_CS);
        int relocOff    = readLE16(header, MZ_RELOC_OFF);

        int fileSize = (int) provider.length();
        int hdrSize  = hdrParas * 16;
        int imageSize = (pages - 1) * 512 + lastPage - hdrSize;
        if (imageSize < 0) imageSize = fileSize - hdrSize;
        
        // Compute load segment (typical: 0x1000)
        int loadSeg = 0x1000;
        
        log.appendMsg(String.format(
            "MZ Header: CS=%04X:%04X  SS:SP=%04X:%04X  " +
            "Image=%04X bytes  Relocs=%d",
            initCS, initIP, initSS, initSP, imageSize, relocCount));

        // Create program segment block
        String blockName = "MZ_IMAGE";
        AddressSpace segSpace = addrFactory.getDefaultAddressSpace();
        
        long physBase = (long) loadSeg * 16;
        MemoryBlock block = memory.createInitializedBlock(
            blockName,
            segSpace.getAddress(physBase),
            provider.getInputStream(hdrSize),
            imageSize,
            (byte) 0x00, // fill
            consumer,
            false  // overlay
        );
        block.setRead(true);
        block.setWrite(false);
        block.setExecute(true);
        
        // Create PSP block
        MemoryBlock pspBlock = memory.createInitializedBlock(
            "PSP",
            segSpace.getAddress((long)(loadSeg) * 16 - PSP_SIZE),
            null, PSP_SIZE, (byte) 0x00, consumer, false
        );
        pspBlock.setRead(true);
        pspBlock.setWrite(true);
        pspBlock.setExecute(false);
        
        // Apply relocations
        if (relocCount > 0 && relocOff > 0) {
            applyMZRelocations(provider, memory, segSpace,
                relocOff, relocCount, loadSeg, monitor);
        }
        
        // Set entry point
        long entryPhys = (long)(loadSeg + initCS) * 16 + initIP;
        program.getSymbolTable().createLabel(
            segSpace.getAddress(entryPhys),
            "ENTRY_POINT",
            program.getGlobalNamespace(),
            null
        );
        
        log.appendMsg(String.format(
            "Entry point at physical 0x%05X (%04X:%04X)",
            entryPhys, loadSeg + initCS, initIP));
    }

    private void applyMZRelocations(ByteProvider provider, Memory memory,
            AddressSpace space, int relocOffset, int relocCount,
            int loadSeg, TaskMonitor monitor) throws IOException {
        
        byte[] relocData = provider.readBytes(relocOffset, relocCount * 4);
        
        for (int i = 0; i < relocCount; i++) {
            if (monitor.isCancelled()) break;
            
            int off = readLE16(relocData, i * 4);
            int seg  = readLE16(relocData, i * 4 + 2);
            
            long phys = (long)(loadSeg + seg) * 16 + off;
            
            // Read the word at relocation site
            byte[] word = new byte[2];
            memory.getBytes(space.getAddress(phys), word);
            int relocVal = (word[0] & 0xFF) | ((word[1] & 0xFF) << 8);
            
            // Add load segment
            relocVal += loadSeg;
            
            // Write back
            memory.setBytes(space.getAddress(phys),
                new byte[]{(byte)(relocVal & 0xFF), (byte)((relocVal >> 8) & 0xFF)});
        }
    }

    // ─── COM Loader ───────────────────────────────────────────

    private void loadCOM(ByteProvider provider, Memory memory,
            AddressFactory addrFactory, Program program,
            MessageLog log, TaskMonitor monitor) throws IOException {
        
        int fileSize = (int) provider.length();
        AddressSpace space = addrFactory.getDefaultAddressSpace();
        
        // COM files load at CS:0100 with 256-byte PSP at CS:0000
        long physBase = (long) COM_LOAD_SEG * 16;
        
        // Create PSP block (0x00-0xFF)
        MemoryBlock psp = memory.createInitializedBlock(
            "PSP", space.getAddress(physBase),
            null, PSP_SIZE, (byte) 0x00, consumer, false
        );
        psp.setRead(true);
        psp.setWrite(true);
        psp.setExecute(false);
        
        // Create code/data block at offset 0x100
        MemoryBlock code = memory.createInitializedBlock(
            "COM_CODE",
            space.getAddress(physBase + PSP_SIZE),
            provider.getInputStream(0),
            fileSize,
            (byte) 0x00, consumer, false
        );
        code.setRead(true);
        code.setWrite(true);
        code.setExecute(true);
        
        // Entry point at CS:0100
        program.getSymbolTable().createLabel(
            space.getAddress(physBase + PSP_SIZE),
            "COM_ENTRY",
            program.getGlobalNamespace(), null
        );
        
        log.appendMsg(String.format(
            "COM file: %d bytes loaded at %04X:0100 (physical 0x%05X)",
            fileSize, COM_LOAD_SEG, physBase + PSP_SIZE));
    }

    // ─── Raw Binary Loader ────────────────────────────────────

    private void loadRAW(ByteProvider provider, Memory memory,
            AddressFactory addrFactory, Program program,
            MessageLog log, TaskMonitor monitor) throws IOException {
        
        int fileSize = (int) provider.length();
        AddressSpace space = addrFactory.getDefaultAddressSpace();
        
        MemoryBlock block = memory.createInitializedBlock(
            "RAW_BINARY",
            space.getAddress(0),
            provider.getInputStream(0),
            fileSize,
            (byte) 0x00, consumer, false
        );
        block.setRead(true);
        block.setWrite(false);
        block.setExecute(true);
        
        log.appendMsg(String.format(
            "Raw binary: %d bytes mapped at 0000:0000", fileSize));
    }

    // ─── Utility ──────────────────────────────────────────────

    private int readLE16(byte[] data, int offset) {
        return (data[offset] & 0xFF) | ((data[offset + 1] & 0xFF) << 8);
    }
}

    @Override
    protected void run() throws Exception {
        println("[*] DOS Binary Analyzer starting...");
        
        // Phase 1: Identify and annotate INT instructions
        annotateInterrupts();
        
        // Phase 2: Create functions at call targets
        createFunctionsAtTargets();
        
        // Phase 3: Annotate known segment register loads
        annotateSegmentLoads();
        
        // Phase 4: Mark data strings
        findDataStrings();

    private void annotateInterrupts() {
        Listing listing = currentProgram.getListing();
        MemoryBlock textBlock = currentProgram.getMemory()
            .getBlock(".text");
        if (textBlock == null) {
            // Try to find any executable block
            for (MemoryBlock b : currentProgram.getMemory().getBlocks()) {
                if (b.isExecute()) { textBlock = b; break; }
            }
        }
        if (textBlock == null) return;
        
        InstructionIterator iter = listing.getInstructions(
            textBlock.getStart(), true);
        
        int count = 0;
        while (iter.hasNext() && !monitor.isCancelled()) {
            Instruction instr = iter.next();
            String mnemonic = instr.getMnemonicString();
            
            if (mnemonic.equals("INT")) {
                Scalar intNum = instr.getScalar(0);
                if (intNum != null) {
                    int num = (int) intNum.getUnsignedValue();
                    annotateInterrupt(instr, num);
                    count++;
                }
            }
        }
        
        println(String.format("[+] Annotated %d interrupt calls", count));
    }

    private void annotateInterrupt(Instruction instr, int intNum) {
        String comment = null;
        
        // Check preceding MOV AH, xx for INT 21h
        if (intNum == 0x21) {
            Integer ahVal = findPrecedingAHValue(instr);
            if (ahVal != null) {
                String svc = DOS_21H_SERVICES.get(ahVal);
                if (svc != null) {
                    comment = String.format("INT 21h AH=%02Xh: %s",
                        ahVal, svc);
                }
            }
            if (comment == null) {
                comment = "INT 21h: DOS Services";
            }
        } else if (intNum == 0x10) {
            Integer ahVal = findPrecedingAHValue(instr);
            if (ahVal != null) {
                String svc = BIOS_10H.get(ahVal);
                if (svc != null) {
                    comment = String.format("INT 10h AH=%02Xh: %s",
                        ahVal, svc);
                }
            }
            if (comment == null) {
                comment = "INT 10h: BIOS Video Services";
            }
        } else {
            comment = String.format("INT %02Xh", intNum);
        }
        
        if (comment != null) {
            instr.setComment(CodeUnit.EOL_COMMENT, comment);
        }
    }

    private Integer findPrecedingAHValue(Instruction intInstr) {
        // Walk backward up to 5 instructions to find MOV AH, imm
        Instruction prev = intInstr;
        for (int i = 0; i < 5; i++) {
            prev = currentProgram.getListing()
                .getInstructionBefore(prev.getAddress());
            if (prev == null) break;
            
            if (prev.getMnemonicString().equals("MOV") &&
                prev.getNumOperands() == 2) {
                String op0 = prev.getDefaultOperandRepresentation(0);
                if (op0.equals("AH") || op0.equals("ah")) {
                    Scalar val = prev.getScalar(1);
                    if (val != null) return (int) val.getUnsignedValue();
                }
            }
        }
        return null;
    }

    private void createFunctionsAtTargets() {
        // Scan for CALL instructions and create function symbols
        Listing listing = currentProgram.getListing();
        InstructionIterator iter = listing.getInstructions(true);
        
        int count = 0;
        while (iter.hasNext() && !monitor.isCancelled()) {
            Instruction instr = iter.next();
            if (instr.getMnemonicString().equals("CALL")) {
                Address target = instr.getAddress(0);
                if (target != null &&
                    currentProgram.getFunctionManager()
                        .getFunctionAt(target) == null) {
                    createFunction(target, "FUN_" + target.toString());
                    count++;
                }
            }
        }
        println(String.format("[+] Created %d function stubs", count));
    }

    private void annotateSegmentLoads() {
        Listing listing = currentProgram.getListing();
        InstructionIterator iter = listing.getInstructions(true);
        
        while (iter.hasNext() && !monitor.isCancelled()) {
            Instruction instr = iter.next();
            String mn = instr.getMnemonicString();
            
            // MOV SEG, imm16 or LES/LDS
            if ((mn.equals("MOV") && instr.getNumOperands() == 2)) {
                String op0 = instr.getDefaultOperandRepresentation(0);
                if (op0.matches("(CS|DS|SS|ES|FS|GS)")) {
                    Scalar val = instr.getScalar(1);
                    if (val != null) {
                        instr.setComment(CodeUnit.EOL_COMMENT,
                            String.format("Set %s = %04Xh (phys base %05Xh)",
                                op0, val.getUnsignedValue(),
                                val.getUnsignedValue() * 16));
                    }
                }
            }
        }
    }

    private void findDataStrings() {
        // Scan for null-terminated ASCII strings in data segments
        MemoryBlock[] blocks = currentProgram.getMemory().getBlocks();
        for (MemoryBlock block : blocks) {
            if (!block.isInitialized() || block.isExecute()) continue;
            // String detection would go here
        }
    }

    private void createFunction(Address addr, String name) {
        try {
            currentProgram.getSymbolTable().createLabel(
                addr, name,
                currentProgram.getGlobalNamespace(),
                SourceType.ANALYSIS);
        } catch (Exception e) {
            // Skip if can't create symbol
        }
    }

☰ 16-bit Register Quick Reference

General Purpose

AX (AH:AL) — Accumulator
BX (BH:BL) — Base (mem addressing)
CX (CH:CL) — Counter (loops, shifts)
DX (DH:DL) — Data (I/O, MUL/DIV)
SI — Source Index (string ops)
DI — Destination Index (string ops)
BP — Base Pointer (stack frames)
SP — Stack Pointer

Segment Registers

CS — Code Segment (instruction fetch)
DS — Data Segment (default data access)
SS — Stack Segment (stack operations)
ES — Extra Segment (string dest, misc)
FS — Extra Segment (386+ only)
GS — Extra Segment (386+ only)

Special

IP — Instruction Pointer (offset in CS)
FLAGS — Status Register
  CF (bit 0) — Carry Flag
  ZF (bit 6) — Zero Flag
  SF (bit 7) — Sign Flag
  OF (bit 11) — Overflow Flag
  DF (bit 10) — Direction Flag
  IF (bit 9) — Interrupt Enable
