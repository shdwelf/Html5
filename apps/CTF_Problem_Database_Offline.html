let activeCat = null;
let sortCol = null;
let sortAsc = true;

// ── INIT ──
function init(){
  renderNav();
  renderTable(DB);
  updateStats(DB);
  populateYearFilter();
}
function getCategories(){
  const order = ["Web","Crypto","Pwn","Reverse","Forensics","OSINT","Misc","A/D"];
  const cats = [...new Set(DB.map(c=>c.cat))];
  return order.filter(o=>cats.includes(o)).concat(cats.filter(c=>!order.includes(c)));
}
function renderNav(){
  const nav = document.getElementById('catNav');
  const cats = getCategories();
  nav.innerHTML = '';
  cats.forEach(c=>{
    const b = document.createElement('button');
    b.textContent = c;
    b.dataset.cat = c;
    b.onclick = ()=>{ activeCat = (activeCat===c ? null : c); filterAndRender(); updateNavButtons(); };
    nav.appendChild(b);
  });
}
function updateNavButtons(){
  document.querySelectorAll('#catNav button').forEach(b=>{
    b.classList.toggle('active', b.dataset.cat === activeCat || (!activeCat && b.dataset.cat===''));
  });
}

// ── FILTERING ──
function getFiltered(){
  let data = DB;
  if(activeCat) data = data.filter(c=>c.cat===activeCat);
  const q = document.getElementById('searchBox').value.toLowerCase();
  if(q) data = data.filter(c=>Object.values(c).some(v=>String(v).toLowerCase().includes(q)));
  const df = document.getElementById('diffFilter').value;
  if(df) data = data.filter(c=>c.diff===df);
  const yf = document.getElementById('yearFilter').value;
  if(yf) data = data.filter(c=>String(c.year)===yf);
  return data;
}
function filterAndRender(){
  const data = getFiltered();
  renderTable(data);
  updateStats(data);
}
function updateStats(data){
  const cats = [...new Set(data.map(c=>c.cat))].length;
  document.getElementById('statsBar').innerHTML =
    `${data.length} challenges`+
    `${cats} categories`+
    `${[...new Set(data.map(c=>c.ctf))].length} CTF events`+
    `Easy: ${data.filter(c=>c.diff==='Easy').length}`+
    `Med: ${data.filter(c=>c.diff==='Med').length}`+
    `Hard: ${data.filter(c=>c.diff==='Hard').length}`;
}

// ── TABLE RENDERING ──
function renderTable(data){
  const head = document.getElementById('tableHead');
  const body = document.getElementById('tableBody');
  const noRes = document.getElementById('noResults');
  if(data.length===0){ body.innerHTML=''; head.innerHTML=''; noRes.style.display='block'; return; }
  noRes.style.display='none';

  const cols = [
    {key:'name',label:'Challenge'},
    {key:'cat',label:'Category'},
    {key:'sub',label:'Sub'},
    {key:'ctf',label:'CTF Event'},
    {key:'year',label:'Year'},
    {key:'diff',label:'Difficulty'},
    {key:'tech',label:'Technique'},
    {key:'tools',label:'Tools'}
  ];

  head.innerHTML = `${cols.map(c=>
    `${c.label} ${sortCol===c.key?(sortAsc?'▲':'▼'):''}`
  ).join('')}`;

  body.innerHTML = data.map(c=>`
    ${esc(c.name)}
    ${esc(c.cat)}${esc(c.sub)}${esc(c.ctf)}
    ${c.year}
    ${c.diff}
    ${esc(c.tech)}${esc(c.tools)}
  `).join('');
}

function esc(s){ return String(s).replace(/&/g,'&').replace(//g,'>'); }

function sortBy(col){
  if(sortCol===col){ sortAsc=!sortAsc; } else { sortCol=col; sortAsc=true; }
  const data = getFiltered();
  data.sort((a,b)=>{
    const va = String(a[col]||''), vb = String(b[col]||'');
    const na = Number(a[col]), nb = Number(b[col]);
    const cmp = isNaN(na)||isNaN(nb) ? va.localeCompare(vb) : na-nb;
    return sortAsc ? cmp : -cmp;
  });
  renderTable(data);
}

  const a = document.createElement('a');
